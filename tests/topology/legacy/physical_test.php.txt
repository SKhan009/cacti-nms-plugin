/** Mutation and isolation checks run only against a disposable copy of Cacti. */

if(empty($o['cacti-root'])||!preg_match('/^topology_canvas_qa_[a-z0-9_]+$/D',$o['database']??'')) exit(2);
require($o['cacti-root'].'/include/cli_check.php');
$database_hostname='localhost';$database_port='3306';$database_default=$o['database'];
if(!db_connect_real('localhost','root','',$database_default,'mysql','3306',1)) exit(2);
require_once(__DIR__.'/../includes/physical.php');
try {
    if(db_fetch_cell('SELECT DATABASE()')!==$o['database']) throw new RuntimeException('QA database mismatch.');
    $admin=(int)read_config_option('admin_user');$_SESSION=array('sess_user_id'=>$admin);
    function pcheck($v,$s){if(!$v)throw new RuntimeException($s);print "PASS: $s\n";}
    function preject($fn,$s){try{$fn();}catch(RuntimeException|InvalidArgumentException $e){print "PASS: $s\n";return;}throw new RuntimeException('Accepted: '.$s);}
    $core=hash('sha256',json_encode(db_fetch_assoc('SELECT * FROM host ORDER BY id')));
    $sources=db_fetch_cell('SELECT COUNT(*) FROM data_local');$graphs=db_fetch_cell('SELECT COUNT(*) FROM graph_local');
    $evidence=db_fetch_assoc('SELECT * FROM plugin_topology_snapshots ORDER BY host_id,protocol');
    foreach(array('port_profiles','ports','cables','positions','scenes') as $table) tp_exec('DROP TABLE IF EXISTS plugin_topology_'.$table);
    tp_schema();tp_schema();
    preject(function(){tp_category_save(0,'Unsafe color','',1,'red;display:none');},'non-hex category color rejected');

    pcheck(db_fetch_cell_prepared('SELECT color FROM plugin_topology_categories WHERE id=?',array($colorCategory))==='#7c3aed','category color saved');
    tp_category_save($colorCategory,'QA color renamed','',1);

    preject(function(){tp_profile_save(0,1,'QA too large','Port',1,129,'RJ45');},'oversized port profile rejected');

    $a=db_fetch_assoc('SELECT * FROM plugin_topology_ports WHERE host_id=9 ORDER BY ordinal');$b=db_fetch_assoc('SELECT * FROM plugin_topology_ports WHERE host_id=10 ORDER BY ordinal');

    $wrong=tp_category_save(0,'QA computers','',1);$wp=tp_profile_save(0,$wrong,'QA wrong','LAN',1,1,'RJ45');

    preject(function()use($a){tp_port_map(3,$a[1]['id'],1);},'duplicate interface mapping rejected');
    preject(function()use($a){tp_port_map(3,$a[1]['id'],99999);},'uncached interface mapping rejected');
    $cable=tp_cable_add(3,$a[0]['id'],$b[0]['id']);
    preject(function()use($a,$b){tp_cable_add(3,$b[1]['id'],$a[0]['id']);},'occupied endpoint rejected in either cable direction');
    preject(function()use($a){tp_cable_add(3,$a[1]['id'],$a[1]['id']);},'self connection rejected');
    preject(function()use($a){tp_cable_add(3,$a[0]['id'],$a[1]['id']);},'same-device cable rejected');
    preject(function(){tp_ports_apply(3,9,0);},'connected physical ports cannot be cleared');
    tp_profile_save($profile,1,'QA two ports','Ethernet',1,3,'RJ45');tp_ports_apply(3,9,$profile);
    $again=db_fetch_assoc('SELECT * FROM plugin_topology_ports WHERE host_id=9 ORDER BY ordinal');
    pcheck(count($again)===3&&$again[0]['id']===$a[0]['id']&&(int)$again[0]['if_index']===1,'profile expansion preserves connected IDs and interface mappings');
    tp_profile_save($profile,1,'QA two ports','Ethernet',2,2,'RJ45');
    preject(function()use($profile){tp_ports_apply(3,9,$profile);},'profile reduction cannot remove connected port');
    pcheck(count(db_fetch_assoc('SELECT id FROM plugin_topology_ports WHERE host_id=9'))===3,'failed profile application rolls back atomically');

    preject(function()use($sibling){tp_assignment_save(9,1,'Ethernet Switch','Core','lldp',$sibling);},'node reassignment requires explicit port cleanup');

    preject(function(){tp_layout_save(3,1,array(array('id'=>9,'x'=>9999,'y'=>0)));},'out-of-bounds layout rejected');

    tp_cable_delete(3,$cable);tp_ports_apply(3,10,0);tp_assignment_save(10,1,'Ethernet Switch','Access','lldp',$sibling);tp_ports_apply($sibling,10,$profile);
    $b2=db_fetch_assoc('SELECT * FROM plugin_topology_ports WHERE host_id=10 ORDER BY ordinal');

    print "PHYSICAL TOPOLOGY TESTS COMPLETE\n";
} catch(Throwable $e) {fwrite(STDERR,$e->getMessage()."\n");exit(1);}
